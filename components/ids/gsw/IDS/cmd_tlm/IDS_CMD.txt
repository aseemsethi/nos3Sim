COMMAND IDS IDS_SET_THRESHOLD_CC <%= CosmosCfsConfig::PROCESSOR_ENDIAN %> "IDS Set Detector Threshold Command"
  APPEND_PARAMETER CCSDS_STREAMID      16 UINT MIN_UINT16 MAX_UINT16 0x18FE "CCSDS Packet Identification" BIG_ENDIAN
  APPEND_PARAMETER CCSDS_SEQUENCE      16 UINT MIN_UINT16 MAX_UINT16 0xC000 "CCSDS Packet Sequence Control" BIG_ENDIAN
  APPEND_PARAMETER CCSDS_LENGTH        16 UINT MIN_UINT16 MAX_UINT16 9      "CCSDS Packet Data Length" BIG_ENDIAN
  APPEND_PARAMETER CCSDS_FC            8  UINT MIN_UINT8 MAX_UINT8 8        "CCSDS Command Function Code"
  APPEND_PARAMETER CCSDS_CHECKSUM      8  UINT MIN_UINT8 MAX_UINT8 0        "CCSDS Command Checksum"
  # Field order mirrors IDS_SetThreshold_cmd_t: Threshold (float) is declared
  # before DetectorId so it lands 4-byte aligned right after CmdHeader with
  # no mid-struct C compiler padding.
  APPEND_PARAMETER THRESHOLD           32 FLOAT MIN_FLOAT32 MAX_FLOAT32 5.0 "Meaning depends on DETECTOR_ID: rate ratio, silence ms, or cmd-error-count spike"
  APPEND_PARAMETER DETECTOR_ID          8 UINT MIN_UINT8 MAX_UINT8 0        "Which detector this threshold applies to"
    STATE RATE       0
    STATE SILENCE    1
    STATE CMD_REJECT 2
  # 3 bytes of trailing padding: the C struct rounds up to a multiple of 4
  # because it contains a float member, same as any other cFS command whose
  # payload size isn't already a multiple of 4. Declared as three individual
  # 8-bit items rather than one 24-bit item - COSMOS's LITTLE_ENDIAN bitfield
  # packer only accepts 8/16/32/64-bit widths, and a bare byte has no
  # endianness ambiguity to begin with.
  APPEND_PARAMETER PAD_1                8 UINT MIN_UINT8 MAX_UINT8 0        "Structure padding, do not set"
  APPEND_PARAMETER PAD_2                8 UINT MIN_UINT8 MAX_UINT8 0        "Structure padding, do not set"
  APPEND_PARAMETER PAD_3                8 UINT MIN_UINT8 MAX_UINT8 0        "Structure padding, do not set"
